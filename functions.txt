2.Check Even or Odd

func checkEvenOdd(num: Int) -> String {
    if num % 2 == 0 {
        return "Even"
    } else {
        return "Odd"
    }
}

print(checkEvenOdd(num: 7))

3. Find Factorial of a Number

func factorial(n: Int) -> Int {
    var result = 1

    if n < 0 {
        return -1
    }

    for i in 1...n {
        result = result * i
    }

    return result
}

print(factorial(n: 5))
